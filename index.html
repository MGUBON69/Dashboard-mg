วิธีติดตั้ง Google Apps Script

คำแนะนำ:

คัดลอกโค้ดด้านล่างนี้ไปวางใน Google Sheet เพื่อเปิด API
  1. เปิด Google Sheet ของคุณ
  2. ไปที่ Extensions > Apps Script
  3. วางโค้ดด้านล่าง → กด Save
  4. กด Deploy → New Deployment
  5. เลือก Web App → Access: Anyone
  6. Deploy แล้วคัดลอก URL มาเชื่อมต่อ
Code.gs 
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "DEALER BK INPUT V2";
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) return jsonResponse({ error: "Sheet not found" });
    var startRow = 4, startCol = 5, numCols = 8;
    var maxRow = sheet.getLastRow(), lastRow = maxRow;
    if (maxRow >= startRow) {
      var colE = sheet.getRange(startRow, startCol, maxRow - startRow + 1, 1).getValues();
      lastRow = colE.filter(String).length + startRow - 1;
    }
    var data = [];
    if (lastRow >= startRow) {
      var rawData = sheet.getRange(startRow, startCol, lastRow - startRow + 1, numCols).getValues();
      for (var i = 0; i < rawData.length; i++) {
        var row = rawData[i];
        if (row[0] !== "") {
          var bs = row[4] ? row[4].toString().trim() : "";
          if (bs.toLowerCase().includes("ไม่ผ่าน")) bs = "ยกเลิก (ไม่ผ่านเงื่อนไข)";
          data.push({
            model: row[0] ? row[0].toString().trim() : "",
            subModel: row[1] ? row[1].toString().trim() : "",
            color: row[2] ? row[2].toString().trim() : "",
            finance: row[3] ? row[3].toString().trim() : "",
            bookingStatus: bs,
            carStatus: row[5] ? row[5].toString().trim() : "",
            note: row[6] ? row[6].toString().trim() : "",
            sc: row[7] ? row[7].toString().trim() : ""
          });
        }
      }
    }
    if (e && e.parameter) {
      if (e.parameter.model) data = data.filter(r => r.model === e.parameter.model);
      if (e.parameter.sc) data = data.filter(r => r.sc === e.parameter.sc);
      if (e.parameter.status) data = data.filter(r => r.bookingStatus === e.parameter.status);
    }
    return jsonResponse({
      totalRecords: data.length,
      lastUpdate: Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss"),
      data: data
    });
  } catch (err) { return jsonResponse({ error: err.message }); }
}
function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
