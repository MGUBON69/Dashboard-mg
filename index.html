วิธีติดตั้ง Google Apps Script ใน Google Sheets

คำแนะนำในการใช้งาน:

คัดลอกโค้ดด้านล่างนี้ไปวางใน Google Sheet ของคุณ เพื่อเปิด API สำหรับดึงข้อมูลมาแสดงบน Dashboard
  1. เปิด Google Sheet BK_M2 ของคุณ
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้ววางโค้ดด้านล่างนี้ลงไป
  4. กดปุ่ม บันทึก (Save)
  5. คลิกปุ่ม ทำให้ใช้งานได้อย่างสะดวก (Deploy) > การทำให้ใช้งานได้อย่างสะดวกรายการใหม่ (New Deployment)
  6. เลือกประเภทเป็น เว็บแอป (Web App)
  7. ตั้งค่า ผู้ที่มีสิทธิ์เข้าถึง (Who has access) เป็น: ทุกคน (Anyone)
  8. กด Deploy แล้วคัดลอก **Web App URL** มาใส่ในช่องเชื่อมต่อของ Dashboard นี้
Code.gs 
function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('DEALER BK INPUT V2');
  
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify({
      error: "ไม่พบชีตชื่อ 'DEALER BK INPUT V2'"
    })).setMimeType(ContentService.MimeType.JSON);
  }
  
  // อ่านข้อมูลตั้งแต่แถวที่ 4 (Row 4) เป็นต้นไป
  var lastRow = sheet.getLastRow();
  if (lastRow < 4) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  
  // อ่านช่วงคอลัมน์ E ถึง L (Column 5 ถึง 12)
  var range = sheet.getRange(4, 5, lastRow - 3, 8);
  var values = range.getValues();
  
  var result = [];
  
  for (var i = 0; i < values.length; i++) {
    var row = values[i];
    var model = row[0] ? row[0].toString().trim() : '';
    
    // ข้ามแถวที่ไม่มีข้อมูลรุ่น
    if (!model) continue;
    
    result.push({
      model: model,                                    // Column E (รุ่น)
      subModel: row[1] ? row[1].toString().trim() : '',// Column F (รุ่นย่อย)
      color: row[2] ? row[2].toString().trim() : '',   // Column G (สี)
      finance: row[3] ? row[3].toString().trim() : '', // Column H (สถาบันการเงิน)
      status: row[4] ? row[4].toString().trim() : '',  // Column I (สถานะการจอง)
      carStatus: row[5] ? row[5].toString().trim() : '',// Column J (สถานะรถ)
      note: row[6] ? row[6].toString().trim() : '',     // Column K (หมายเหตุ)
      sc: row[7] ? row[7].toString().trim() : ''        // Column L (SC)
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
