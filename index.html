วิธีติดตั้ง Google Apps Script ใน Google Sheets

คำแนะนำในการใช้งาน:

คัดลอกโค้ดด้านล่างนี้ไปวางใน Google Sheet ของคุณ เพื่อเปิด API สำหรับดึงข้อมูลมาแสดงบน Dashboard
  1. เปิด Google Sheet BK_M2 ของคุณ
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้ววางโค้ดด้านล่างนี้ลงไป
  4. กดปุ่ม บันทึก (Save)
  5. คลิกปุ่ม ทำให้ใช้งานได้อย่างสะดวก (Deploy) > การทำให้ใช้งานได้อย่างสะดวกรายการใหม่ (New Deployment)
  6. เลือกประเภทเป็น เว็บแอป (Web App)
  7. ตั้งค่า ผู้ที่มีสิทธิ์เข้าถึง (Who has access) เป็น: ทุกคน (Anyone)
  8. กด Deploy แล้วคัดลอก **Web App URL** มาใส่ในช่องเชื่อมต่อของ Dashboard นี้

หมายเหตุสำคัญ:

  • ต้อง Deploy ใหม่ (New Deployment) ทุกครั้งที่แก้ไขโค้ด
  • ตรวจสอบว่า Sheet ชื่อว่า DEALER BK INPUT V2 อยู่แล้ว
  • ข้อมูลต้องเริ่มที่แถวที่ 4 และอยู่ในคอลัมน์ E ถึง L
Code.gs 
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "DEALER BK INPUT V2";
    var sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      return jsonResponse({ error: "Sheet not found: " + sheetName });
    }
    
    var startRow = 4;
    var startCol = 5;
    var numCols = 8;
    var maxRow = sheet.getLastRow();
    var lastRow = maxRow;
    
    // หา lastRow ของคอลัมน์ E โดยเฉพาะ
    if (maxRow >= startRow) {
      var colE = sheet.getRange(startRow, startCol, maxRow - startRow + 1, 1).getValues();
      lastRow = colE.filter(String).length + startRow - 1;
    }
    
    var data = [];
    if (lastRow >= startRow) {
      var numRows = lastRow - startRow + 1;
      var rawData = sheet.getRange(startRow, startCol, numRows, numCols).getValues();
      
      for (var i = 0; i < rawData.length; i++) {
        var row = rawData[i];
        if (row[0] !== "") {
          data.push({
            model: row[0] ? row[0].toString().trim() : "",
            subModel: row[1] ? row[1].toString().trim() : "",
            color: row[2] ? row[2].toString().trim() : "",
            finance: row[3] ? row[3].toString().trim() : "",
            bookingStatus: row[4] ? row[4].toString().trim() : "",
            carStatus: row[5] ? row[5].toString().trim() : "",
            note: row[6] ? row[6].toString().trim() : "",
            sc: row[7] ? row[7].toString().trim() : ""
          });
        }
      }
    }
    
    // รองรับการกรองผ่าน Query Parameter
    if (e && e.parameter) {
      if (e.parameter.model) {
        data = data.filter(function(r) { return r.model === e.parameter.model; });
      }
      if (e.parameter.sc) {
        data = data.filter(function(r) { return r.sc === e.parameter.sc; });
      }
      if (e.parameter.status) {
        data = data.filter(function(r) { return r.bookingStatus === e.parameter.status; });
      }
    }
    
    var output = {
      totalRecords: data.length,
      lastUpdate: Utilities.formatDate(new Date(), "Asia/Bangkok", "yyyy-MM-dd HH:mm:ss"),
      data: data
    };
    
    return jsonResponse(output);
    
  } catch (err) {
    return jsonResponse({ error: err.message, stack: err.stack });
  }
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
